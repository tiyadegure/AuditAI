VulnerableToken.sol 
pragma solidity ^0.8.0;

contract VulnerableToken {
    mapping(address => uint256) public balances;

    function transfer(address to, uint256 amount) public {
        // Bug: No overflow check!
        balances[msg.sender] -= amount;
        balances[to] += amount;
    }

    function withdraw() public {
        // Bug: Reentrancy vulnerability!
        (bool success, ) = msg.sender.call{value: balances[msg.sender]}("");
        require(success);
        balances[msg.sender] = 0;
    }
}

💻 Code Examples

See AuditAI in action with real code

CLI Usage 
# Scan a single contract
$ audit-ai scan Contract.sol

# Scan with full options
$ audit-ai scan Contract.sol \
    --detector all \
    --attest \
    --output report.md

# Use knowledge base
$ audit-ai scan Contract.sol \
    --knowledge-base
Python SDK 
from audit_ai import Scanner

# Initialize scanner
scanner = Scanner(
    detectors="all",
    attest=True
)

# Scan contract
results = scanner.scan(
    "Contract.sol"
)

# Print findings
for finding in results.findings:
    print(f"[{finding.severity}] {finding.name}")
